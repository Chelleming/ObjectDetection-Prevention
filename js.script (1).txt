/* ============ Web Audio: 3D spatial ============ */
let soundOn = true;
let masterVol = 1.0;
let actx = null;
function ctx(){ if(!actx) actx = new (window.AudioContext||window.webkitAudioContext)(); return actx; }

function updateListener(px,py,pz,fx,fz){
  if(!actx) return;
  const l = actx.listener;
  if(l.positionX){
    l.positionX.setValueAtTime(px, actx.currentTime);
    l.positionY.setValueAtTime(py, actx.currentTime);
    l.positionZ.setValueAtTime(pz, actx.currentTime);
    l.forwardX.setValueAtTime(fx, actx.currentTime);
    l.forwardY.setValueAtTime(0, actx.currentTime);
    l.forwardZ.setValueAtTime(fz, actx.currentTime);
    l.upX.setValueAtTime(0, actx.currentTime);
    l.upY.setValueAtTime(1, actx.currentTime);
    l.upZ.setValueAtTime(0, actx.currentTime);
  } else if(l.setPosition){
    l.setPosition(px,py,pz);
    l.setOrientation(fx,0,fz, 0,1,0);
  }
}

function makePanner(x,y,z){
  const c = ctx();
  const panner = c.createPanner();
  panner.panningModel = 'HRTF';
  panner.distanceModel = 'inverse';
  panner.refDistance = 2;
  panner.maxDistance = 70;
  panner.rolloffFactor = 1.1;
  if(panner.positionX){
    panner.positionX.setValueAtTime(x, c.currentTime);
    panner.positionY.setValueAtTime(y, c.currentTime);
    panner.positionZ.setValueAtTime(z, c.currentTime);
  } else if(panner.setPosition){
    panner.setPosition(x,y,z);
  }
  return panner;
}

function ping3D(x,y,z, opts){
  if(!soundOn) return;
  opts = opts || {};
  const c = ctx();
  const osc = c.createOscillator();
  const gain = c.createGain();
  const panner = makePanner(x,y,z);
  osc.type = opts.wave || 'sine';
  osc.frequency.setValueAtTime(opts.freq || 520, c.currentTime);
  if(opts.freqTo !== undefined){
    osc.frequency.exponentialRampToValueAtTime(Math.max(opts.freqTo,20), c.currentTime + (opts.dur||0.28));
  }
  const vol = (opts.volume !== undefined ? opts.volume : 0.5) * masterVol;
  gain.gain.setValueAtTime(0.0001, c.currentTime);
  gain.gain.exponentialRampToValueAtTime(Math.max(vol,0.001), c.currentTime+0.02);
  gain.gain.exponentialRampToValueAtTime(0.0001, c.currentTime + (opts.dur||0.28));
  osc.connect(gain); gain.connect(panner); panner.connect(c.destination);
  osc.start(); osc.stop(c.currentTime + (opts.dur||0.28) + 0.03);
}
// double-pulse "siren" for moving/patrol hazards -- distinct timbre from static ping
function siren3D(x,y,z, opts){
  if(!soundOn) return;
  opts = opts || {};
  ping3D(x,y,z, Object.assign({wave:'sawtooth', freq:(opts.freq||300), dur:0.12, volume:(opts.volume||0.4)}, {}));
  setTimeout(()=>{ ping3D(x,y,z, Object.assign({wave:'sawtooth', freq:(opts.freq||300)*1.25, dur:0.12, volume:(opts.volume||0.4)*0.85}, {})); }, 130);
}
function twinkle3D(x,y,z, opts){
  if(!soundOn) return;
  opts = opts || {};
  ping3D(x,y,z, {wave:'sine', freq:opts.freq||900, freqTo:(opts.freq||900)*1.6, dur:0.22, volume:opts.volume||0.35});
}
function tone2D(freq, dur, volume, wave){
  if(!soundOn) return;
  const c = ctx();
  const osc = c.createOscillator();
  const gain = c.createGain();
  osc.type = wave || 'triangle';
  osc.frequency.setValueAtTime(freq, c.currentTime);
  gain.gain.setValueAtTime(0.0001, c.currentTime);
  gain.gain.exponentialRampToValueAtTime((volume||0.25)*masterVol, c.currentTime+0.02);
  gain.gain.exponentialRampToValueAtTime(0.0001, c.currentTime+(dur||0.3));
  osc.connect(gain); gain.connect(c.destination);
  osc.start(); osc.stop(c.currentTime+(dur||0.3)+0.03);
}

function clearGroup(arr){ arr.forEach(o=> scene.remove(o.mesh)); arr.length = 0; }
function randRange(a,b){ return a + Math.random()*(b-a); }

function farEnough(p, list, minDist){
  return !list.some(o => o.mesh.position.distanceTo(p) < minDist);
}

function spawnHazards(staticCount, patrolCount){
  clearGroup(hazards);
  const startPos = player.position.clone();
  const goalPos = goalMesh.position.clone();
  let attempts = 0;
  while(hazards.length < staticCount && attempts < staticCount*25){
    attempts++;
    const p = new THREE.Vector3(randRange(-FIELD_HALF+3,FIELD_HALF-3), 0.6, randRange(-FIELD_HALF+3,FIELD_HALF-3));
    if(p.distanceTo(startPos) < 6 || p.distanceTo(goalPos) < 5) continue;
    if(!farEnough(p, hazards, 3.2)) continue;
    const size = randRange(0.9,1.4);
    const mesh = new THREE.Mesh(new THREE.BoxGeometry(size,size,size), staticHazardMat);
    mesh.position.copy(p); mesh.castShadow = true;
    scene.add(mesh);
    hazards.push({mesh, radius:size*0.62+0.55, type:'static'});
  }
  attempts = 0;
  let placed = 0;
  while(placed < patrolCount && attempts < patrolCount*25){
    attempts++;
    const ax = randRange(-FIELD_HALF+3,FIELD_HALF-3), az = randRange(-FIELD_HALF+3,FIELD_HALF-3);
    const a = new THREE.Vector3(ax,0.6,az);
    if(a.distanceTo(startPos) < 6 || a.distanceTo(goalPos) < 5) continue;
    if(!farEnough(a, hazards, 3.5)) continue;
    const angle = Math.random()*Math.PI*2;
    const len = randRange(4,9);
    const b = new THREE.Vector3(a.x+Math.cos(angle)*len, 0.6, a.z+Math.sin(angle)*len);
    b.x = Math.max(-FIELD_HALF+2, Math.min(FIELD_HALF-2, b.x));
    b.z = Math.max(-FIELD_HALF+2, Math.min(FIELD_HALF-2, b.z));
    const size = randRange(0.8,1.1);
    const mesh = new THREE.Mesh(new THREE.BoxGeometry(size,size,size), patrolHazardMat);
    mesh.position.copy(a); mesh.castShadow = true;
    scene.add(mesh);
    hazards.push({mesh, radius:size*0.62+0.55, type:'patrol', a, b, t:Math.random()*Math.PI*2, speed:randRange(0.6,1.1)});
    placed++;
  }
}

  const startPos = player.position.clone();
  const goalPos = goalMesh.position.clone();
  let attempts = 0;
  while(orbs.length < orbCount && attempts < orbCount*25){
    attempts++;
    const p = new THREE.Vector3(randRange(-FIELD_HALF+3,FIELD_HALF-3), 0.9, randRange(-FIELD_HALF+3,FIELD_HALF-3));
    if(p.distanceTo(startPos) < 4 || p.distanceTo(goalPos) < 4) continue;
    if(!farEnough(p, hazards, 2.2) || !farEnough(p, orbs, 2.5)) continue;
    const mesh = new THREE.Mesh(new THREE.SphereGeometry(0.32,10,10), orbMat);
    mesh.position.copy(p); mesh.castShadow = true;
    scene.add(mesh);
    orbs.push({mesh, value:10});
  }
  if(Math.random() < shieldChance){
    attempts = 0;
    while(attempts < 25){
      attempts++;
      const p = new THREE.Vector3(randRange(-FIELD_HALF+3,FIELD_HALF-3), 0.9, randRange(-FIELD_HALF+3,FIELD_HALF-3));
      if(p.distanceTo(startPos) < 4 || p.distanceTo(goalPos) < 4) continue;
      if(!farEnough(p, hazards, 2.5) || !farEnough(p, orbs, 2.5)) continue;
      const mesh = new THREE.Mesh(new THREE.TetrahedronGeometry(0.42,0), shieldMat);
      mesh.position.copy(p); mesh.castShadow = true;
      scene.add(mesh);
      powerups.push({mesh, kind:'shield'});
      break;
    }
  }
}

/* ---- Google-Maps-style relative directions ----
   The player always faces "north" (-Z), so left/right/ahead/behind
   can be computed directly from world-space offsets, the way a
   turn-by-turn nav app gives directions relative to your heading. */
function bearingDeg(from, to){
  const dx = to.x - from.x, dz = to.z - from.z;
  const worldAngle = (Math.atan2(dx, -dz) * 180/Math.PI + 360) % 360; // 0 = world north
  const headingDeg = (heading * 180/Math.PI + 360) % 360;
  return ((worldAngle - headingDeg) % 360 + 360) % 360; // 0=ahead, 90=right, 180=behind, 270=left
}
function navPhrase(angle){
  if(angle < 20 || angle > 340) return "straight ahead";
  if(angle < 70) return "ahead, slightly right";
  if(angle < 110) return "on your right";
  if(angle < 160) return "behind you, to the right";
  if(angle < 200) return "directly behind you";
  if(angle < 250) return "behind you, to the left";
  if(angle < 290) return "on your left";
  return "ahead, slightly left";
}
function turnSide(angle){
  if(angle >= 340 || angle < 20) return "straight ahead";
  return angle < 180 ? "on your right" : "on your left";
}
function oppositeSide(side){
  if(side.includes("right")) return "left";
  if(side.includes("left")) return "right";
  return Math.random() < 0.5 ? "left" : "right";
}

function nearestHazard(){
  if(hazards.length===0) return null;
  let best=null, bd=Infinity;
  hazards.forEach(h=>{
    const d = h.mesh.position.distanceTo(player.position);
    if(d<bd){bd=d; best=h;}
  });
  return {hazard:best, dist:bd};
}

function doScan(manual){
  hazards.forEach(h=>{
    const d = h.mesh.position.distanceTo(player.position);
    if(d < 45){
      const vol = Math.max(0.05, 0.55 - d*0.01);
      const freq = 200 + Math.max(0,(20-d))*18;
      if(h.type === 'patrol'){
        siren3D(h.mesh.position.x, 0.8, h.mesh.position.z, {freq, volume:vol});
      } else {
        ping3D(h.mesh.position.x, 0.8, h.mesh.position.z, {freq, dur:0.3, volume:vol, wave:'sine'});
      }
    }
  });
  orbs.forEach(o=>{
    const d = o.mesh.position.distanceTo(player.position);
    if(d < 30) twinkle3D(o.mesh.position.x, 0.9, o.mesh.position.z, {volume:Math.max(0.06,0.32-d*0.008)});
  });
  powerups.forEach(p=>{
    const d = p.mesh.position.distanceTo(player.position);
    if(d < 30) ping3D(p.mesh.position.x, 0.9, p.mesh.position.z, {wave:'triangle', freq:520, freqTo:1040, dur:0.35, volume:Math.max(0.08,0.4-d*0.008)});
  });
  const gd = goalMesh.position.distanceTo(player.position);
  ping3D(goalMesh.position.x, 1.1, goalMesh.position.z, {freq: 880, dur:0.4, volume: Math.max(0.08,0.5-gd*0.008), wave:'triangle'});

  if(!manual){
    // proactive, Maps-style heads-up: only speak when a hazard is close and roughly ahead,
    // so it feels like "in 8 meters, turn left" rather than constant chatter
    const n = nearestHazard();
    if(n && n.dist < 14){
      const angle = bearingDeg(player.position, n.hazard.mesh.position);
      const ahead = angle < 100 || angle > 260;
      if(ahead){
        const side = turnSide(angle);
        const kind = n.hazard.type === 'patrol' ? "moving hazard" : "hazard";
        if(n.dist < 4){
          speak(kind.charAt(0).toUpperCase()+kind.slice(1) + " right ahead! Turn " + oppositeSide(side) + " now.", true);
        } else {
          speak("In " + Math.round(n.dist) + " meters, " + kind + " " + side + ". Keep " + oppositeSide(side) + ".");
        }
      }
    }
    return;
  }

  if(manual){
    const n = nearestHazard();
    const goalAngle = bearingDeg(player.position, goalMesh.position);
    const goalDir = navPhrase(goalAngle);

    if(gd < 1.6){ speak("You have arrived at your destination."); if(tutorialMode) tutorialTips.scanned = true; return; }

    if(verbosity === 'short'){
      if(!n){ speak("Clear ahead. Destination " + Math.round(gd) + " meters, " + goalDir + "."); }
      else{
        const side = turnSide(bearingDeg(player.position, n.hazard.mesh.position));
        speak("Hazard " + Math.round(n.dist) + " meters, " + side + ". Destination " + Math.round(gd) + " meters, " + goalDir + ".");
      }
    } else {
      if(!n){
        speak("No hazards nearby. Continue toward your destination, it's " + Math.round(gd) + " meters, " + goalDir + ".");
      } else {
        const side = turnSide(bearingDeg(player.position, n.hazard.mesh.position));
        const kind = n.hazard.type === 'patrol' ? "moving hazard" : "hazard";
        if(n.dist < 5){
          speak("Caution, a " + kind + " is just ahead, " + side + ". Turn " + oppositeSide(side) + " to avoid it. Your destination is " + Math.round(gd) + " meters, " + goalDir + ".", true);
        } else {
          speak("In " + Math.round(n.dist) + " meters, a " + kind + " " + side + ". Keep " + oppositeSide(side) + ". Destination is " + Math.round(gd) + " meters, " + goalDir + ".");
        }
      }
    }
    if(tutorialMode) tutorialTips.scanned = true;
  }
}

function knockback(from, to){
  const dir = new THREE.Vector3().subVectors(to, from).setY(0).normalize();
  to.add(dir.multiplyScalar(2.2));
  to.x = Math.max(-FIELD_HALF+1, Math.min(FIELD_HALF-1, to.x));
  to.z = Math.max(-FIELD_HALF+1, Math.min(FIELD_HALF-1, to.z));
}

  speak("Game over. Final score " + score + ". Press Enter to play again.", true);
  chatLog("Final score: " + score + ".", "Game Over");
}

function beginLevel(n){
  level = n;
  hazardCount = Math.min(4 + (n-1)*2, 16);
  patrolFraction = Math.min(0.15*(n-1), 0.6);
  pulseInterval = Math.max(3.2 - (n-1)*0.15, 1.4);
  const patrolCount = Math.round(hazardCount*patrolFraction);
  const staticCount = hazardCount - patrolCount;
  player.position.set(0,0,FIELD_HALF-4);
  heading = 0;
  placeGoal();
  spawnHazards(staticCount, patrolCount);
  spawnPickups(Math.min(2+Math.floor(n/2),5), 0.35);
  updateHUD();
}

function levelUp(){
  flash('win');
  score += 25;
  speak("You have arrived at your destination! Level " + (level+1) + ". More hazards ahead.", true);
  chatLog("Reached the destination. Starting level " + (level+1) + ".", "Level Up");
  beginLevel(level+1);
}

function setDifficultyBase(name, btn){
  document.querySelectorAll('#easyBtn,#medBtn,#hardBtn').forEach(b=>b.setAttribute('aria-pressed','false'));
  btn.setAttribute('aria-pressed','true');
  window.__difficulty = name;
  speak(name + " difficulty selected.");
}
window.__difficulty = 'easy';

function applyDifficultyToLevel(){
  const d = window.__difficulty;
  if(d==='easy'){ hazardCount = Math.max(3, hazardCount-2); pulseInterval += 0.4; }
  if(d==='hard'){ hazardCount = hazardCount+3; pulseInterval = Math.max(1.1, pulseInterval-0.5); }
}

function startTutorial(){
  tutorialMode = true;
  score = 0; lives = 3; invuln = 0; shieldTimer = 0; pulseTimer = 0;
  tutorialTips = {moved:false, scanned:false};
  player.position.set(0,0,FIELD_HALF-4);
  heading = 0;
  placeGoal();
  goalMesh.position.set(0, 1.1, 6); // close, straight ahead
  spawnHazards(1, 0);
  hazards[0] && hazards[0].mesh.position.set(0, 0.6, -2); // directly on the straight path
  spawnPickups(1, 0);
  pulseInterval = 3.5;
  running = true;
  centerMsg.hidden = true;
  updateHUD();
  chatLog("Practice round: reach your destination, dodge the one hazard, grab the orb if you like.", "Tutorial");
  speak("Practice round. Press the up arrow or W to head toward your destination, and use the arrow keys or WASD to move around. I'll give you turn-by-turn directions as you go.", true);
}

function finishTutorial(){
  tutorialMode = false;
  chatLog("Practice complete. Starting level 1.", "Tutorial");
  speak("Nicely done. Starting the full game now.", true);
  applyDifficultyToLevel();
  score = 0;
  beginLevel(1);
}

function startGame(fromSkip){
  try{ ctx().resume(); }catch(e){}
  if(fromSkip){
    tutorialMode = false;
    score=0; lives=3; invuln=0; shieldTimer=0; pulseTimer=0;
    hazardCount = 5; patrolFraction = 0;
    applyDifficultyToLevel();
    running = true;
    centerMsg.hidden = true;
    beginLevel(1);
    chatLog("Starting the full game.", "System");
    speak("Game started. Walk forward to head toward your destination. Press Space any time for directions.", true);
  } else {
    startTutorial();
  }
}

function togglePause(force){
  if(!running && force !== false) return;
  paused = (force !== undefined) ? force : !paused;
  pauseMsg.hidden = !paused;
  if(paused){ speak("Paused.", true); }
  else { speak("Resumed."); }
}

let lastT = performance.now();
function animate(now){
  requestAnimationFrame(animate);
  const dt = Math.min((now-lastT)/1000, 0.05);
  lastT = now;

  goalMesh.rotation.y += dt*1.2;
  goalMesh.position.y = 1.1 + Math.sin(now*0.002)*0.15;
  orbs.forEach((o,i)=>{ o.mesh.rotation.y += dt*2; o.mesh.position.y = 0.9 + Math.sin(now*0.003+i)*0.12; });
  powerups.forEach((p)=>{ p.mesh.rotation.y += dt*3; p.mesh.rotation.x += dt*1.5; });

  if(running && !paused){
    // patrol hazard movement
    hazards.forEach(h=>{
      if(h.type === 'patrol'){
        h.t += dt*h.speed;
        const s = (Math.sin(h.t)+1)/2;
        h.mesh.position.lerpVectors(h.a, h.b, s);
        h.mesh.rotation.y += dt*2;
      }
    });

    let mx=0, mz=0;
    if(keys.up) mz -= 1;
    if(keys.down) mz += 1;
    if(keys.left) mx -= 1;
    if(keys.right) mx += 1;
    const moving = (mx!==0 || mz!==0);
    if(moving){
      const len = Math.hypot(mx,mz);
      mx/=len; mz/=len;
      player.position.x += mx*MOVE_SPEED*dt;
      player.position.z += mz*MOVE_SPEED*dt;
      player.position.x = Math.max(-FIELD_HALF+1, Math.min(FIELD_HALF-1, player.position.x));
      player.position.z = Math.max(-FIELD_HALF+1, Math.min(FIELD_HALF-1, player.position.z));
      walkT += dt*8;
      const parts = player.userData.parts;
      const swing = Math.sin(walkT)*0.55;
      parts.legL.rotation.x = swing; parts.legR.rotation.x = -swing;
      parts.armL.rotation.x = -swing; parts.armR.rotation.x = swing;
    } else {
      const parts = player.userData.parts;
      parts.legL.rotation.x *= 0.8; parts.legR.rotation.x *= 0.8;
      parts.armL.rotation.x *= 0.8; parts.armR.rotation.x *= 0.8;
    }

    updateListener(player.position.x, 1.2, player.position.z, 0, -1);

    // tutorial guided tips
    if(tutorialMode){
      if(!tutorialTips.moved && now % 6000 < 20){ /* passive, handled via timers below */ }
    }

    pulseTimer += dt;
    if(pulseTimer >= pulseInterval){
      pulseTimer = 0;
      doScan(false);
    }

    if(shieldTimer > 0){
      shieldTimer -= dt;
      player.userData.shieldRing.visible = true;
      player.userData.shieldRing.rotation.z += dt*3;
      if(shieldTimer <= 0){ player.userData.shieldRing.visible = false; speak("Shield expired."); updateHUD(); }
    }

    if(invuln>0) invuln -= dt;
    if(invuln<=0 && shieldTimer<=0){
      for(const h of hazards){
        const d = h.mesh.position.distanceTo(player.position);
        if(d < h.radius){
          lives -= 1;
          invuln = 1.4;
          flash('hit');
          tone2D(140,0.35,0.35,'sawtooth');
          const side = turnSide(bearingDeg(player.position, h.mesh.position));
          speak("Ouch! Hazard " + side + ". Lives left: " + Math.max(lives,0) + ".", true);
          chatLog("Collided with a hazard (" + dir + "). Lives: " + Math.max(lives,0) + ".", "Hit");
          knockback(h.mesh.position, player.position);
          updateHUD();
          if(lives<=0 && !tutorialMode){ endGame(); }
          else if(lives<=0 && tutorialMode){ lives = 3; player.position.set(0,0,FIELD_HALF-4); heading = 0; speak("Let's try that again.", true); updateHUD(); }
          break;
        }
      }
    } else if(invuln<=0 && shieldTimer>0){
      for(const h of hazards){
        const d = h.mesh.position.distanceTo(player.position);
        if(d < h.radius){
          invuln = 0.6;
          tone2D(700,0.15,0.2,'sine');
          speak("Shield absorbed a hit.");
          knockback(h.mesh.position, player.position);
          break;
        }
      }
    }

    // orb pickup
    for(let i=orbs.length-1;i>=0;i--){
      const o = orbs[i];
      if(o.mesh.position.distanceTo(player.position) < 0.9){
        score += o.value;
        flash('pick');
        twinkle3D(o.mesh.position.x,0.9,o.mesh.position.z,{volume:0.4});
        chatLog("Collected a bonus orb. +" + o.value + " points.", "Pickup");
        scene.remove(o.mesh);
        orbs.splice(i,1);
        updateHUD();
      }
    }
    // shield pickup
    for(let i=powerups.length-1;i>=0;i--){
      const p = powerups[i];
      if(p.mesh.position.distanceTo(player.position) < 0.9){
        shieldTimer = 8;
        flash('pick');
        ping3D(p.mesh.position.x,0.9,p.mesh.position.z,{wave:'triangle',freq:500,freqTo:1200,dur:0.4,volume:0.4});
        speak("Shield activated! You're briefly immune to hazards.", true);
        chatLog("Picked up a shield power-up.", "Pickup");
        scene.remove(p.mesh);
        powerups.splice(i,1);
        updateHUD();
      }
    }

    const gd2 = goalMesh.position.distanceTo(player.position);
    if(gd2 < 1.6){
      if(tutorialMode) finishTutorial();
      else levelUp();
    }
  }

  updateCamera();
  renderer.render(scene, camera);
}
requestAnimationFrame(animate);

});
